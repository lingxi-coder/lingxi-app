#[test]
fn registry_exposes_exactly_40_builtin_tools_v0_5_0() {
    use lingxi_test_harness::parity::load_fixture;
    use std::collections::BTreeSet;

    #[derive(serde::Deserialize)]
    struct Fixture {
        total: usize,
        by_category: std::collections::BTreeMap<String, Vec<String>>,
    }

    let fx: Fixture = load_fixture("registry_40_tools");
    assert_eq!(fx.total, 40, "fixture asserts total=40");

    // Cross-check: sum of by_category lengths equals 40.
    let summed: usize = fx.by_category.values().map(Vec::len).sum();
    assert_eq!(summed, 40, "by_category lengths sum to 40 (got {summed})");

    // Derive expected sorted-name set from the fixture.
    let expected: BTreeSet<String> = fx
        .by_category
        .values()
        .flat_map(|v| v.iter().cloned())
        .collect();
    assert_eq!(expected.len(), 40, "expected names are unique");

    // Build the registry by invoking the real register_all_builtin_tools entry point
    // with the test-harness MockBuiltinToolContext (the same context M4-01..08 use).
    let ctx = lingxi_test_harness::mocks::mock_builtin_tool_context();

    lingxi_tools::builtin::register_all_builtin_tools(&mut reg, ctx);

    let actual: BTreeSet<String> = reg.iter().map(|t| t.name().to_string()).collect();
    assert_eq!(
        actual, expected,
        "registry contents diverge from fixture; symmetric diff: missing-from-registry={:?}, extra-in-registry={:?}",
        expected.difference(&actual).collect::<Vec<_>>(),
        actual.difference(&expected).collect::<Vec<_>>(),
    );
    assert_eq!(reg.iter().count(), 40, "registry cardinality is 40");
}

//! Parity driver: every builtin tool that produces variable-length output
//! routes it through `lingxi_tools::output_truncation::truncate` (or its

    use lingxi_test_harness::mocks::mock_builtin_tool_context;

    let ctx = mock_builtin_tool_context();

    let exempt = exempt_tools();

    // Map ToolName ("NotebookEdit") → src file ("notebook_edit.rs"). M4-01 lock:
    // builtin file = snake-case of tool name.
    fn snake_of(name: &str) -> String {
        // Inline lightweight snake_case: prepend `_` before each uppercase
        // after the first char, then lowercase the whole thing.
        let mut out = String::with_capacity(name.len() + 4);
        for (i, ch) in name.chars().enumerate() {
            if i > 0 && ch.is_uppercase() {
                out.push('_');
            }
            for c in ch.to_lowercase() {
                out.push(c);
            }
        }
        out
    }

        .parent().unwrap()  // crates/test-harness → crates
