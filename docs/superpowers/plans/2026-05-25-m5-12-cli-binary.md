- [ ] **Step 3: Lock the `--json` output schema.**

  When `--json` is passed, the CLI emits **one JSON object per stdout line** (NDJSON / JSONL) so tools can parse incrementally. Locked event shapes:

```json
// turn lifecycle:
{"event":"turn_start","ts":"2026-05-26T10:00:00Z","session_id":"<uuid>"}
{"event":"text","ts":"<rfc3339>","content":"<chunk>"}
{"event":"tool_call","ts":"<rfc3339>","tool":"<name>","input":{...}}
{"event":"tool_result","ts":"<rfc3339>","tool":"<name>","result":{...}}
{"event":"turn_end","ts":"<rfc3339>","stop_reason":"end_turn","cost":{"total_usd":0.0042,"input_tokens":100,"output_tokens":50}}

// slash command outputs (when typed in a turn):
{"event":"command_output","ts":"<rfc3339>","name":"clear","display":"Conversation cleared."}

// errors:
{"event":"error","ts":"<rfc3339>","code":"<runtime|api|argv>","message":"<text>"}
```

  Schema rules:
  - Every line is a JSON object (no arrays, no trailing commas).
  - First field is always `event` (one of the 6 above).
  - Second field is always `ts` (RFC 3339 UTC with `Z` suffix, second precision).
  - Subsequent fields depend on event type.
  - Lines are `\n`-terminated (LF only; no CRLF even on Windows — `--json` is a machine-readable mode).
  - Schema additions in M6+ will gate behind new event names; existing 6 are locked.

```rust
use clap::Parser;

/// AI coding assistant — runs a single turn or REPL
#[derive(Debug, Parser, Clone)]
#[command(name = "lingxi-cli", version, about, long_about = None)]

    ///
    /// When absent (and `--resume` is not set), enters REPL mode (M5-13).
    pub prompt: Option<String>,

    /// Print mode: exit after first end_turn

    pub print: bool,

    /// Resume a previous session by UUID (or interactive picker if absent)
    #[arg(long = "resume", value_name = "ID", num_args = 0..=1, default_missing_value = "")]
    pub resume: Option<String>,

    /// Override the active model (e.g. claude-opus-4-7)
    #[arg(long = "model", value_name = "NAME")]
    pub model: Option<String>,

    /// Change to this directory before initialising
    #[arg(long = "cwd", value_name = "DIR")]

    /// Disable streaming SSE; use batched messages_create instead
    #[arg(long = "no-stream")]
    pub no_stream: bool,

    /// Emit machine-readable NDJSON to stdout (one event per line)
    #[arg(long = "json")]
    pub json: bool,

    /// Enable verbose logging to stderr

    pub debug: bool,
}

impl Argv {
    /// Parse from any iterable of `OsString` (used by integration tests).
    #[must_use]
    pub fn from_iter<I, T>(iter: I) -> Result<Self, clap::Error>
    where
        I: IntoIterator<Item = T>,
        T: Into<std::ffi::OsString> + Clone,
    {
        Self::try_parse_from(iter)
    }

    /// True iff the binary should start the REPL.

    #[must_use]
    pub fn is_repl_mode(&self) -> bool {
        let no_prompt = self.prompt.as_deref().map_or(true, |s| s.trim().is_empty());
        no_prompt && self.resume.is_none()
    }
}
```

```rust
use std::path::{Path, PathBuf};

#[derive(Debug, thiserror::Error)]
pub enum CwdError {
    #[error("--cwd path does not exist: {0}")]
    NotFound(PathBuf),
    #[error("--cwd path is not a directory: {0}")]
    NotADirectory(PathBuf),
    #[error("could not change directory to {0}: {1}")]
    ChdirFailed(PathBuf, std::io::Error),
}

/// Apply `--cwd` if set: validate + `std::env::set_current_dir`.
pub fn apply_cwd(cwd: Option<&Path>) -> Result<(), CwdError> {
    let Some(p) = cwd else { return Ok(()); };
    if !p.exists() {
        return Err(CwdError::NotFound(p.to_path_buf()));
    }
    if !p.is_dir() {
        return Err(CwdError::NotADirectory(p.to_path_buf()));
    }
    std::env::set_current_dir(p)
        .map_err(|e| CwdError::ChdirFailed(p.to_path_buf(), e))?;
    Ok(())
}
```

```rust
pub async fn run_oneshot(argv: &Argv, runtime: &Runtime, sink: &dyn OutputSink) -> i32 {
    let prompt = argv.prompt.clone().unwrap_or_default();
    if prompt.trim().is_empty() {
        eprintln!("lingxi-cli: empty prompt");
        return exit_codes::ARGV_ERROR;
    }

    // 1. Check if the input is a slash command (M5-09 dispatcher).
    if prompt.starts_with('/') {
        return run_slash_command(&prompt, runtime, sink).await;
    }

    // 2. Otherwise drive the orchestrator turn loop.
    match runtime.orchestrator.run_one_shot(&prompt).await {
        Ok(()) => exit_codes::SUCCESS,
        Err(e) => {
            sink.error("runtime", &e.to_string()).await;
            exit_codes::RUNTIME_ERROR
        }
    }
}

async fn run_slash_command(input: &str, runtime: &Runtime, sink: &dyn OutputSink) -> i32 {
    use lingxi_platform_api::{SlashCommandDispatcher, SlashDispatchResult};
    match runtime.dispatcher.dispatch(input).await {
        SlashDispatchResult::Handled { display } => {
            sink.text(&display).await;
            sink.text("\n").await; // ensure trailing newline
            exit_codes::SUCCESS

        SlashDispatchResult::Unknown { display, .. } => {
            sink.text(&display).await;
            sink.text("\n").await;
            exit_codes::RUNTIME_ERROR

        SlashDispatchResult::NotASlashCommand => {
            // Cannot happen — we only enter this branch when the input
            // starts with '/'. Treat as runtime error.
            sink.error("runtime", "not a slash command (internal error)").await;
            exit_codes::RUNTIME_ERROR

    }
```

```rust
pub async fn run_resume(argv: &Argv, runtime: &Runtime, sink: &dyn OutputSink) -> i32 {
    let cwd = std::env::current_dir().unwrap_or_default();
    let arg = argv.resume.as_deref().unwrap_or("");
    let session_id = match resolve_session_id(arg, &cwd).await {
        Ok(Some(id)) => id,
        Ok(None) => {
            // User cancelled the interactive picker.
            return exit_codes::SUCCESS;
        }
        Err(e) => {
            sink.error("runtime", &e.to_string()).await;
            return exit_codes::RUNTIME_ERROR;
        }
    };

    // Hand off to orchestrator's resume entrypoint.
    match runtime.orchestrator.resume_session(session_id).await {
        Ok(()) => {
            // After resume, if a positional prompt was also supplied, run
            // one more turn; otherwise enter REPL (deferred to M5-13).
            if let Some(p) = &argv.prompt {
                if !p.trim().is_empty() {
                    match runtime.orchestrator.run_one_shot(p).await {
                        Ok(()) => exit_codes::SUCCESS,
                        Err(e) => {
                            sink.error("runtime", &e.to_string()).await;
                            exit_codes::RUNTIME_ERROR
                        }
                    }
                } else {
                    eprintln!("lingxi-cli: resumed; REPL not yet wired (M5-13)");
                    exit_codes::NOT_IMPLEMENTED
                }
            } else {
                eprintln!("lingxi-cli: resumed; REPL not yet wired (M5-13)");
                exit_codes::NOT_IMPLEMENTED
            }
        }
        Err(e) => {
            sink.error("runtime", &e.to_string()).await;
            exit_codes::RUNTIME_ERROR
        }
    }
}

async fn resolve_session_id(arg: &str, cwd: &std::path::Path) -> Result<Option<uuid::Uuid>, LoaderError> {
    if arg.is_empty() {
        // Interactive picker over the 5 most-recent sessions in this cwd.
        let sessions = list_recent_sessions(cwd, 5).await?;
        if sessions.is_empty() {
            return Err(LoaderError::EmptyDirectory);
        }
        select_session_interactive(&sessions, &mut tokio::io::stdin(), &mut tokio::io::stderr())
            .await
            .map(|opt| opt.map(|m| m.uuid))
    } else {
        // Direct UUID lookup.
        let id = uuid::Uuid::parse_str(arg)
            .map_err(|_| LoaderError::SessionNotFound { arg: arg.to_string() })?;
        Ok(Some(id))
    }
}
```

```rust
//! build.rs — capture the short git SHA at compile time for /version.
//!
//! Falls back to "unknown" when:
//! - not inside a git repo
//! - git command not available
//! - any other failure
//!
//! The captured value is read by `lingxi-commands/src/builtin/version.rs`
//! via `option_env!("LINGXI_GIT_SHA_SHORT")`.

use std::process::Command;

fn main() {
    let sha = Command::new("git")
        .args(["rev-parse", "--short", "HEAD"])
        .output()
        .ok()
        .and_then(|o| if o.status.success() {
            Some(String::from_utf8_lossy(&o.stdout).trim().to_string())
        } else {
            None
        })
        .unwrap_or_else(|| "unknown".to_string());

    println!("cargo:rustc-env=LINGXI_GIT_SHA_SHORT={sha}");
    // Re-run if HEAD moves.
    println!("cargo:rerun-if-changed=../../.git/HEAD");
    println!("cargo:rerun-if-changed=../../.git/refs/heads");
    println!("cargo:rerun-if-changed=build.rs");
}
```
